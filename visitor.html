async function startCamera() { try { // Eseye kamera devan an dabò localStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" }, audio: false }); console.log("✅ Kamera devan louvri"); return true; } catch (err) { console.warn("Erè kamera devan:", err); // Si kamera devan pa mache, eseye nenpòt kamera try { localStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false }); console.log("✅ Yon lòt kamera louvri"); return true; } catch (err2) { console.warn("Erè kamera:", err2); return false; } } }
